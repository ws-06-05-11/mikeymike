# ─── Hulpfuncties ─────────────────────────────────────────────────────────────

function Get-LatestCsv ([string]$Folder) {
    if (-not (Test-Path $Folder)) { return $null }
    Get-ChildItem -Path $Folder -Filter "*.csv" |
        Sort-Object LastWriteTime -Descending |
        Select-Object -First 1
}

function ConvertTo-HtmlRows {
    param(
        $Data,
        [string[]]$Columns,
        [hashtable]$StatusColumns = @{}
    )

    if (-not $Data -or $Data.Count -eq 0) { return "" }

    $sb = [System.Text.StringBuilder]::new()
    foreach ($row in $Data) {
        [void]$sb.Append("<tr>")
        foreach ($col in $Columns) {
            $val     = if ($null -ne $row.$col) { $row.$col.ToString() } else { "" }
            $escaped = $val -replace '&','&amp;' -replace '<','&lt;' -replace '>','&gt;'
            $class   = ""

            if ($StatusColumns.ContainsKey($col)) {
                $type = $StatusColumns[$col]
                if ($type -eq "status") {
                    if     ($val -match "Start|Running") { $class = ' class="ok"'   }
                    elseif ($val -match "Stop|Error")    { $class = ' class="stop"' }
                    elseif ($val -match "Unknown|Niet")  { $class = ' class="warn"' }
                }
                elseif ($type -eq "notempty") {
                    if ($val -ne "-" -and $val -ne "")  { $class = ' class="warn"' }
                }
            }

            [void]$sb.Append("<td$class>$escaped</td>")
        }
        [void]$sb.Append("</tr>")
    }
    return $sb.ToString()
}

# ─── HTTP Server starten ───────────────────────────────────────────────────────

$url = "http://localhost:$Port/"

try {
    $listener = [System.Net.HttpListener]::new()
    $listener.Prefixes.Add($url)
    $listener.Start()
}
catch {
    Write-Error "Kon server niet starten op poort $Port.`nIs de poort al in gebruik? Probeer een andere poort via -Port 9090`n$_"
    exit 1
}

Write-Host ""
Write-Host "  ╔═══════════════════════════════════════════════════╗" -ForegroundColor Cyan
Write-Host "  ║   IT Operations Dashboard — Webserver             ║" -ForegroundColor Cyan
Write-Host "  ╠═══════════════════════════════════════════════════╣" -ForegroundColor Cyan
Write-Host "  ║   URL    : http://localhost:$Port/                   ║" -ForegroundColor Cyan
Write-Host "  ║   Data   : $OutputBase                        ║" -ForegroundColor Cyan
Write-Host "  ║   Stop   : Ctrl+C                                  ║" -ForegroundColor Cyan
Write-Host "  ╚═══════════════════════════════════════════════════╝" -ForegroundColor Cyan
Write-Host ""

try {
    while ($listener.IsListening) {
        $context  = $listener.GetContext()
        $request  = $context.Request
        $response = $context.Response
        $path     = $request.Url.AbsolutePath

        if ($path -eq "/" -or $path -eq "/dashboard") {
            Write-Host "  [$(Get-Date -Format 'HH:mm:ss')]  $($request.RemoteEndPoint)  →  Dashboard" -ForegroundColor DarkGray
            try {
                $html   = Build-DashboardHtml
                $buffer = [System.Text.Encoding]::UTF8.GetBytes($html)
                $response.ContentType     = "text/html; charset=utf-8"
                $response.ContentLength64 = $buffer.Length
                $response.StatusCode      = 200
                $response.OutputStream.Write($buffer, 0, $buffer.Length)
            }
            catch {
                $err    = "Fout bij genereren dashboard: $_"
                $buffer = [System.Text.Encoding]::UTF8.GetBytes($err)
                $response.StatusCode      = 500
                $response.ContentType     = "text/plain; charset=utf-8"
                $response.ContentLength64 = $buffer.Length
                $response.OutputStream.Write($buffer, 0, $buffer.Length)
                Write-Warning $err
            }
        }
        else {
            $msg    = "404 - Niet gevonden: $path"
            $buffer = [System.Text.Encoding]::UTF8.GetBytes($msg)
            $response.StatusCode      = 404
            $response.ContentLength64 = $buffer.Length
            $response.OutputStream.Write($buffer, 0, $buffer.Length)
        }

        $response.OutputStream.Close()
    }
}
finally {
    $listener.Stop()
    Write-Host "`n  Server gestopt." -ForegroundColor Yellow
}
