#Requires -Version 5.1
<#
.SYNOPSIS
    Slaat SMTP-inloggegevens versleuteld op als PSCredential-bestand.

.DESCRIPTION
    Gebruikt Windows DPAPI (via Export-Clixml) om de credentials te versleutelen.
    Het bestand kan ALLEEN worden gelezen door dezelfde Windows-gebruiker op
    dezelfde machine. Voer dit script daarom uit als het account dat de
    Scheduled Task uitvoert.

    Na opslaan geeft het script het pad terug dat je doorgeeft aan
    -SmtpCredentialFile in Get-WsusUpdateReport.ps1.

.PARAMETER SmtpUser
    Gebruikersnaam (e-mailadres) voor SMTP-authenticatie.

.PARAMETER OutputFile
    Pad voor het credential-bestand (standaard: naast dit script).

.EXAMPLE
    .\Save-WsusSmtpCredential.ps1 -SmtpUser "wsus@bedrijf.nl"

.EXAMPLE
    .\Save-WsusSmtpCredential.ps1 -SmtpUser "wsus@bedrijf.nl" `
        -OutputFile "C:\Scripts\wsus_smtp.cred"

.NOTES
    Als de Scheduled Task als SYSTEM draait, voer dit script dan uit via:
        psexec.exe -s powershell.exe -File ".\Save-WsusSmtpCredential.ps1" -SmtpUser "..."
    Zo wordt het credential-bestand versleuteld onder het SYSTEM-account.

    Aanbevolen: gebruik een dedicated service-account in plaats van SYSTEM,
    zodat je gewoon als dat account kunt inloggen om dit script te draaien.
#>

[CmdletBinding()]
param(
    [Parameter(Mandatory = $true)]
    [string]$SmtpUser,

    [string]$OutputFile = (Join-Path (Split-Path $MyInvocation.MyCommand.Path) "wsus_smtp.cred")
)

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

Write-Host ""
Write-Host "  SMTP Credential opslaan" -ForegroundColor Cyan
Write-Host "  Gebruiker : $SmtpUser"
Write-Host "  Bestand   : $OutputFile"
Write-Host ""
Write-Host "  Voer het SMTP-wachtwoord in (wordt niet zichtbaar):" -ForegroundColor Yellow

$securePassword = Read-Host -AsSecureString "  Wachtwoord"

if ($securePassword.Length -eq 0) {
    Write-Error "Geen wachtwoord ingevoerd. Afgebroken."
    exit 1
}

$credential = [System.Management.Automation.PSCredential]::new($SmtpUser, $securePassword)

# Sla op als versleuteld XML-bestand (DPAPI, per gebruiker + machine)
$outputDir = Split-Path $OutputFile -Parent
if ($outputDir -and -not (Test-Path $outputDir)) {
    New-Item -ItemType Directory -Path $outputDir -Force | Out-Null
}

$credential | Export-Clixml -Path $OutputFile -Force

# Beperk toegang: alleen eigenaar mag het bestand lezen
$acl = Get-Acl -Path $OutputFile
$acl.SetAccessRuleProtection($true, $false)
$currentUser = [System.Security.Principal.WindowsIdentity]::GetCurrent().Name
$rule = [System.Security.AccessControl.FileSystemAccessRule]::new(
    $currentUser,
    [System.Security.AccessControl.FileSystemRights]::FullControl,
    [System.Security.AccessControl.AccessControlType]::Allow
)
$acl.SetAccessRule($rule)
Set-Acl -Path $OutputFile -AclObject $acl

Write-Host ""
Write-Host "  Credential opgeslagen en beveiligd." -ForegroundColor Green
Write-Host ""
Write-Host "  Gebruik dit pad bij het aanmaken van de Scheduled Task:" -ForegroundColor Yellow
Write-Host "  -SmtpCredentialFile `"$OutputFile`""
Write-Host ""
Write-Host "  Let op: dit bestand is versleuteld voor account '$currentUser'" -ForegroundColor DarkYellow
Write-Host "  op machine '$($env:COMPUTERNAME)'. Kopieer het niet naar andere machines." -ForegroundColor DarkYellow
