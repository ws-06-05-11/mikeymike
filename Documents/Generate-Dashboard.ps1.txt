#Requires -Version 5.1
<#
.SYNOPSIS
    Genereert een HTML dashboard van de CSV output van de 4 scripts.

.PARAMETER OutputBase

[CmdletBinding()]

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

function Get-LatestCsv ([string]$Folder) {
    if (-not (Test-Path $Folder)) { return $null }
    Get-ChildItem -Path $Folder -Filter "*.csv" |
        Sort-Object LastWriteTime -Descending |
        Select-Object -First 1
}

function ConvertTo-HtmlRows {
    param(
        $Data,
        [string[]]$Columns,
        [hashtable]$StatusColumns = @{}
    )

    if (-not $Data -or $Data.Count -eq 0) { return "" }

    $sb = [System.Text.StringBuilder]::new()
    foreach ($row in $Data) {
        [void]$sb.Append("<tr>")
        foreach ($col in $Columns) {
            $val = if ($null -ne $row.$col) { $row.$col.ToString() } else { "" }
            $escaped = $val -replace '&','&amp;' -replace '<','&lt;' -replace '>','&gt;'

            $class = ""
            if ($StatusColumns.ContainsKey($col)) {
                $type = $StatusColumns[$col]
                if ($type -eq "status") {
                    if     ($val -match "Start|Running")   { $class = ' class="ok"'   }
                    elseif ($val -match "Stop|Error")      { $class = ' class="stop"' }
                    elseif ($val -match "Unknown|Niet")    { $class = ' class="warn"' }
                } elseif ($type -eq "notempty") {
                    if ($val -ne "-" -and $val -ne "") { $class = ' class="warn"' }
                }
            }

            [void]$sb.Append("<td$class>$escaped</td>")
        }
        [void]$sb.Append("</tr>")
    }
    return $sb.ToString()
}
